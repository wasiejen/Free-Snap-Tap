# compaction tool for agent

### Context Management Rule
When fetching raw web pages, scraping large documents, or reading long outputs, summarize the essential findings for your task, then immediately execute the `compact_memory` tool to prune raw tokens before proceeding to the next step.

await context.client.session.compact({
  path: { id: context.sessionId },
  body: {
    keep: {
      tokens: 10000 // Retain only the last 10,000 tokens during emergency compaction

    }
  }
});

Having the agent call a compact_session tool on demand—and pairing it with an automatic plugin hook that catches context window overflow errors—creates a resilient self-healing setup.
Part 1: Custom Tool for Agent Self-Compaction

Exposing a dedicated tool allows your agent to proactively clean its history after ingesting heavy web payloads.
TypeScript

import { tool } from "@opencode/sdk";

export const compactTool = tool({
  name: "compact_memory",
  description: "Triggers immediate context compaction to summarize history and free up token space after reading large files or web pages.",
  parameters: {},
  execute: async (_args, context) => {
    try {
      await context.client.session.compact({
        path: { id: context.sessionId },
        body: {
          // Optional override parameters if supported by your API version
          keepTokens: 15000 
        }
      });
      return "Context successfully compacted. Old raw outputs summarized.";
    } catch (err: any) {
      return `Compaction request failed: ${err.message}`;
    }
  }
});

# compaction on context limit

Part 2: Plugin Hook for Automatic Error Recovery

When a request fails due to context overflow ("exceeds the available context size"), OpenCode catches the error through the session.error or llm.error plugin hooks.

You can intercept this error, trigger session.compact(), and retry or resume execution automatically:
TypeScript

export const ContextRecoveryPlugin = {
  name: "context-recovery-plugin",
  
  // Intercept errors during generation or tool processing
  "session.error": async (error: any, context: any) => {
    const errorMsg = String(error?.message || error);
    
    // Check for context size overflow string patterns
    if (
      errorMsg.includes("exceeds the available context size") ||
      errorMsg.includes("context length exceeded") ||
      errorMsg.includes("prompt is too long")
    ) {
      console.warn(`[ContextRecovery] Overflown context detected in session ${context.sessionId}. Triggering emergency compaction...`);

      try {
        // Trigger forced compaction
        await context.client.session.compact({
          path: { id: context.sessionId }
        });

        console.log(`[ContextRecovery] Emergency compaction complete. Session recovered.`);
        
        // Return recovery flag or allow session retry depending on SDK loop setup
        return { handled: true, action: "retry" };
      } catch (compactErr) {
        console.error(`[ContextRecovery] Emergency compaction failed:`, compactErr);
      }
    }
  }
};

/**
 * 1. CUSTOM TOOL: Agent Self-Compaction Tool
 */
export const compactMemoryTool = tool({
  name: "compact_memory",
  description: "Triggers immediate session compaction to free context space after ingesting heavy data.",
  parameters: {},
  execute: async (_args, context) => {
    try {
      await context.client.session.compact({
        path: { id: context.sessionId }
      });

      // Returning this string ensures the post-compaction response contains the directive
      return `Context successfully compacted.\n\n${COMPACTION_RELOAD_DIRECTIVE}`;
    } catch (err: any) {
      return `Compaction request failed: ${err.message}`;
    }
  }
});

/**
 * 2. PLUGIN HOOK: Emergency Context Recovery
 */
export const EmergencyCompactionPlugin = {
  name: "emergency-compaction-plugin",

  "session.error": async (error: any, context: any) => {
    const errorMsg = String(error?.message || error);

    // Detect context window overflow errors
    if (
      errorMsg.includes("exceeds the available context size") ||
      errorMsg.includes("context length exceeded") ||
      errorMsg.includes("prompt is too long")
    ) {
      try {
        // Trigger forced compaction
        await context.client.session.compact({
          path: { id: context.sessionId }
        });

        // Inject the directive message directly into the session history after recovery
        await context.client.session.promptAsync({
          path: { id: context.sessionId },
          body: {
            parts: [
              {
                type: "text",
                text: COMPACTION_RELOAD_DIRECTIVE,
                synthetic: true
              }
            ]
          }
        });

        return { handled: true, action: "retry" };
      } catch (compactErr) {
        console.error("Emergency compaction failed:", compactErr);
      }
    }
  }
};
