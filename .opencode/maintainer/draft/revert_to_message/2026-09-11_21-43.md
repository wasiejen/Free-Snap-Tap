generic example who this could work


```js
export default {
  description: "Reverts the session history to the state immediately preceding the last tool call.",
  parameters: {
    type: "object",
    properties: {
      sessionId: {
        type: "string",
        description: "Target session ID. Defaults to active session."
      }
    }
  },

  execute: async (args: any, context: any) => {
    try {
      const targetSessionId = args.sessionId || context.sessionId;

      // 1. Fetch current session state and messages
      const sessionData = await context.client.session.get({
        path: { id: targetSessionId }
      });

      const messages = sessionData.messages || [];
      if (messages.length === 0) {
        return "Cannot revert: Session history is empty.";
      }

      // 2. Find the last message that triggered or contains a tool call
      // Working backward from the end of history
      let targetMessageId: string | null = null;

      for (let i = messages.length - 1; i >= 0; i--) {
        const msg = messages[i];
        
        // If we find a user message or an assistant message PRIOR to the last tool execution:
        if (msg.role === "user" || (msg.role === "assistant" && !msg.toolCalls)) {
          targetMessageId = msg.id;
          break;
        }
      }

      if (!targetMessageId) {
        return "Could not determine a valid pre-tool message checkpoint.";
      }

      // 3. Truncate/Revert history up to the target message ID
      await context.client.session.revert({
        path: { id: targetSessionId },
        body: {
          messageId: targetMessageId
        }
      });

      return `Session successfully reverted to message '${targetMessageId}' (before the last tool execution).`;
    } catch (err: any) {
      return `Revert failed: ${err.message}`;
    }
  }
};
```
